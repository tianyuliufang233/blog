# 脚本内容feishu.sh：
```
#!/bin/bash
set -e
#echo $1
body_start='{ "msg_type": "text","content": { "text": "'
body_end='" }}'
body="${body_start} $1 ${body_end}"
echo ${body}
#echo curl -v -X POST -H "Content-Type: application/json"  -d  '{ "msg_type": "text","content": { "text": "'$1'" } }'  $2
curl -i -X POST -H "Content-Type: application/json"  -d  "${body}"  $2
```
